# 命令行多轮对话示例的配置文件
# 键值对写法：KEY=VALUE，井号开头是注释。留空的项用默认值。

# 百炼的 API Key（形如 sk-xxxx）。留空则读取环境变量 DASHSCOPE_API_KEY
API_KEY=
API_KEY_ENV=DASHSCOPE_API_KEY

# O 系兼容入口（脚本会自动拼 /chat/completions）
BASE_URL=https://dashscope.aliyuncs.com/compatible-mode/v1
MODEL=qwen3.7-plus

# 采样参数
TEMPERATURE=0.7
MAX_TOKENS=1024
TIMEOUT=60

# 输出约束：text（自由聊天）/ schema（JSON Schema 硬约束）/ json_object（只保证是 JSON）
STRUCT_MODE=text

# 流式输出：仅 STRUCT_MODE=text 时生效；schema 模式必须等完整 JSON
STREAM=true

# 首轮 system 提示词
SYSTEM_PROMPT=你是一个简洁、准确的中文助手。回答尽量短，必要时才展开。

# 最多保留多少条历史消息（user + assistant 合计，system 不计）
MAX_HISTORY_MESSAGES=20

# 每轮是否打印 token 用量
SHOW_USAGE=true

# 退出时是否把对话存成 txt
SAVE_TRANSCRIPT=false

# --- 以下为脚本自动补充的键，可直接改 ---
# 抽取事实用的模型，留空则跟 MODEL 一致
EXTRACT_MODEL=qwen3.8-flash
# 档案注入上限（字符）
PROFILE_MAX_CHARS=300
# 每轮是否自动抽取档案条目
EXTRACT_MEMORY=true
# 抽取置信度门槛，低于它不写入
MIN_CONFIDENCE=0.7
# 每次召回几段归档原文
RECALL_BLOCKS=2
# 召回原文总长度上限
RECALL_MAX_CHARS=600
# 召回相似度门槛（查询被覆盖比例）
RECALL_MIN_SCORE=0.35
# 记忆目录，留空则脚本同级 memory/
MEMORY_DIR=

# --- 以下为脚本自动补充的键，可直接改 ---
# 抽取时把已有主题词回灌给模型，减少同义新词（建议开着）
TOPIC_HINT=true

# --- 以下为脚本自动补充的键，可直接改 ---
# 召回方式：auto=每轮由模型决策；fixed=每轮都查
RECALL_MODE=auto
# 召回决策用的模型，留空则跟 EXTRACT_MODEL 一致
DECIDE_MODEL=
# 召回决策的输出上限
DECIDE_MAX_TOKENS=160

# --- 以下为脚本自动补充的键，可直接改 ---
# 把每次出网请求的报文存到 logs/，方便核对发了几次请求
PACKET_LOG=true
# 报文日志目录，留空则脚本同级 logs/
LOG_DIR=

# --- 以下为脚本自动补充的键，可直接改 ---
# 判同一事实的字面重合度门槛，低于它才走覆盖
DUP_SIMILARITY=0.85

# --- 以下为脚本自动补充的键，可直接改 ---
# 智能体循环：开=多步调工具；关=单步请求
AGENT_MODE=true
# 一轮最多走几步
MAX_STEPS=6
# 单次工具返回给模型的字符上限
TOOL_RESULT_MAX_CHARS=1200

# --- 以下为脚本自动补充的键，可直接改 ---
# 工具目录，留空则脚本同级 tools/
TOOLS_DIR=
# 技能目录，留空则脚本同级 skills/
SKILLS_DIR=
# 是否把技能名和说明列进 system 提示词
SKILLS_ENABLED=true

# --- 以下为脚本自动补充的键，可直接改 ---
# 文件工具的工作目录，留空则脚本同级 workspace/
FS_ROOT= 
# 单次读取的字节上限
FS_READ_MAX_BYTES=262144
# 单次写入的字符上限
FS_WRITE_MAX_CHARS=20000
# 写文件前：ask=询问；allow=直接写；deny=拒绝
FS_WRITE_MODE=ask
# 允许写的扩展名
FS_WRITE_EXTS=.md,.txt,.json,.py,.csv,.log,.html,.yml,.yaml
# 覆盖前是否备份到 FS_ROOT/.backups/
FS_BACKUP=true
